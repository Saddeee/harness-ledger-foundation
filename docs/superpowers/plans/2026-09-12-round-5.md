```ts
export type TimelineNode = {
  id: string;                      // `${kind}:${rowId}`
  kind: "version" | "external_change" | "decision" | "skill" | "verdict";
  at: string;                      // ISO/SQLite timestamp used for ordering (version: created_at; external_change: fetched_at; decision: reviewed_at/decided_at; skill: fetched_at; verdict: created_at)

  summary: string | null;          // "+3 −1 lines" | "1 rule added" | the rule text (decision/verdict) | null
  content: string | null;          // full Knowledge/skill text for version/external_change/skill; rule text for decision/verdict

  restored_from: number | null;    // version id
  improvement_id: number | null;   // decision/verdict → the Suggestions detail

};
```
